cmake_minimum_required(VERSION 3.20)
project(MoonNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../.build/native-deps")
find_package(OpenCV REQUIRED CONFIG PATHS "${CMAKE_CURRENT_SOURCE_DIR}/../.build/opencv-native" NO_DEFAULT_PATH)
file(STRINGS "${OpenCV_DIR}/CMakeCache.txt" OPENCV_HOME REGEX "^CMAKE_HOME_DIRECTORY:INTERNAL=")
string(REGEX REPLACE "^CMAKE_HOME_DIRECTORY:INTERNAL=" "" OPENCV_SOURCE_ROOT "${OPENCV_HOME}")
find_library(MOON_JPEG NAMES libjpeg-turbo PATHS "${OpenCV_DIR}/3rdparty/lib" NO_DEFAULT_PATH REQUIRED)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_CURL OFF CACHE BOOL "" FORCE)
set(USE_PTHREADS ON CACHE BOOL "" FORCE)
set(TESTS OFF CACHE BOOL "" FORCE)
set(UTILS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/cfitsio-4.6.4" cfitsio EXCLUDE_FROM_ALL)
file(GLOB ERFA_SOURCES "${DEPS}/erfa-2.0.1/src/*.c")
list(FILTER ERFA_SOURCES EXCLUDE REGEX "/(t_erfa.*|erfaversion)\\.c$")
add_library(moon_erfa STATIC ${ERFA_SOURCES})
target_include_directories(moon_erfa PUBLIC "${DEPS}/erfa-2.0.1/src")
add_library(MoonNative SHARED MoonNative.cpp Recognition.cpp EyepieceEdges.cpp)
set(ARCHIVE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/libarchive")
target_include_directories(MoonNative PUBLIC include PRIVATE ${OpenCV_INCLUDE_DIRS}
    "${DEPS}/cspice/include" "${DEPS}/cfitsio-4.6.4" ${ARCHIVE_INCLUDE_DIR}
    "${OPENCV_SOURCE_ROOT}/3rdparty/libjpeg-turbo/src" "${OpenCV_DIR}/3rdparty/libjpeg-turbo")
target_link_libraries(MoonNative PRIVATE ${OpenCV_LIBS} ${MOON_JPEG} "${DEPS}/cspice/lib/cspice.a" moon_erfa cfitsio z archive)
set_target_properties(MoonNative PROPERTIES MACOSX_RPATH ON INSTALL_NAME_DIR "@rpath"
    BUILD_WITH_INSTALL_NAME_DIR ON LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.build/native/lib")
target_compile_options(MoonNative PRIVATE -Wall -Wextra)
